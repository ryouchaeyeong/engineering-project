// 아두이노 진동 모터 제어 코드
const int VIBE_PIN = 9; // PWM 지원 핀 (3, 5, 6, 9, 10, 11)

void setup() {
  Serial.begin(9600);
  pinMode(VIBE_PIN, OUTPUT);
  digitalWrite(VIBE_PIN, LOW);
}

void loop() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();

    if (input == "1") {
      digitalWrite(VIBE_PIN, HIGH);
    } else if (input == "0") {
      digitalWrite(VIBE_PIN, LOW);
    } else if (input.startsWith("P")) {
      // PWM 세기 조절 (예: "P255", "P128")
      int pwmVal = input.substring(1).toInt();
      analogWrite(VIBE_PIN, constrain(pwmVal, 0, 255));
    }
  }
}
